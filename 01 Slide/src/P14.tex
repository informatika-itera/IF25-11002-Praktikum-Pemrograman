\begin{frame}[fragile]{Rekursi bercabang dan rekursi pada array}

int fib(int n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
}
int jumlah(int a[], int n) {
    if (n == 0) return 0;
    return a[n - 1] + jumlah(a, n - 1);
}
int main() {
    int a[4] = {5, 8, 10, 2};
