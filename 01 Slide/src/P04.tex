\begin{frame}[fragile]{switch untuk pilihan bernilai tetap}
\framesubtitle{switch-case}
\begin{columns}[T]
\begin{column}{0.55\textwidth}

switch (pilihan) {
    case 1: cout << "Nasi goreng"; break;
    case 2: cout << "Mie ayam"; break;

    default: cout << "Menu tidak ada";
}

\end{Kode}
\end{column}
